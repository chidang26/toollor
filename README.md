
### TERMUX INSTALLATION >
```python
apt update && apt upgrade -y
pkg install git
pkg install python
pip install requests
rm -rf FB-BRUTE
git clone --depth=1 https://github.com/U7P4L-IN/FB-BRUTE.git
cd FB-BRUTE
chmod +x *
python3 FB.py or python FB2.py
```

```python

>> cd FB-BRUTE
>> python3 -m pip install requests bs4

>> python3 FB.py or python FB2.py
```
